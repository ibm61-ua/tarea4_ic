cmake_minimum_required(VERSION 3.15)
project(knn2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Opciones de profiling con gprof (-pg -g -O0 por defecto, o alternativa con -O2)
option(ENABLE_PROFILING "Habilitar soporte de profiling con gprof (-pg)" ON)
option(PROFILING_O2 "Usar optimizacion -O2 para profiling en lugar de -O0" OFF)

if(ENABLE_PROFILING)
  if(PROFILING_O2)
    message(STATUS "Profiling habilitado con -O2: -pg -g -O2")
    add_compile_options(-pg -g -O2)
    add_link_options(-pg)
  else()
    message(STATUS "Profiling habilitado con -O0: -pg -g -O0")
    add_compile_options(-pg -g -O0)
    add_link_options(-pg)
  endif()
endif()

add_executable(cv main.cc crossValidation.cc mnist.cc knn.cc sample.cc)
target_include_directories(cv PRIVATE ${PROJECT_SOURCE_DIR})

include(CTest)
if(BUILD_TESTING)
  add_executable(knn2_tests tests.cc sample.cc knn.cc mnist.cc crossValidation.cc)
  add_test(NAME knn2_tests COMMAND knn2_tests)
endif()
